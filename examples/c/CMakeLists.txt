# Build and run the C and C++ examples against the wickra-verify C ABI.
#
#   cargo build --release -p wickra-verify-c
#   cmake -S examples/c -B examples/c/build
#   cmake --build examples/c/build --config Release
#   ctest --test-dir examples/c/build -C Release --output-on-failure
#
# WICKRA_VERIFY_LIB_DIR defaults to the workspace target/release directory and
# can be overridden for out-of-tree builds.

cmake_minimum_required(VERSION 3.15)
project(wickra_verify_examples C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(WICKRA_VERIFY_LIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../target/release"
    CACHE PATH "Directory containing the built wickra-verify C ABI library")

set(WICKRA_VERIFY_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../bindings/c/include")

find_library(WICKRA_VERIFY_LIB
    NAMES wickra_verify libwickra_verify
    HINTS "${WICKRA_VERIFY_LIB_DIR}"
    REQUIRED)

add_executable(verify verify.c)
target_include_directories(verify PRIVATE "${WICKRA_VERIFY_INCLUDE}")
target_link_libraries(verify PRIVATE "${WICKRA_VERIFY_LIB}")

add_executable(verifypp verify.cpp)
target_include_directories(verifypp PRIVATE "${WICKRA_VERIFY_INCLUDE}")
target_link_libraries(verifypp PRIVATE "${WICKRA_VERIFY_LIB}")

# The golden corpus, globbed at configure time and written into a header, so the
# test covers a claim added to the corpus without editing the test.
# CONFIGURE_DEPENDS makes the build system re-glob when the directory changes.
set(GOLDEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../golden")
file(GLOB GOLDEN_CLAIM_FILES CONFIGURE_DEPENDS "${GOLDEN_DIR}/claims/*.json")
file(GLOB GOLDEN_DATA_FILES CONFIGURE_DEPENDS "${GOLDEN_DIR}/data/*.csv")
set(GOLDEN_CLAIM_ENTRIES "")
foreach(claim_file ${GOLDEN_CLAIM_FILES})
    get_filename_component(claim_name "${claim_file}" NAME)
    string(APPEND GOLDEN_CLAIM_ENTRIES "    \"${claim_name}\",\n")
endforeach()
set(GOLDEN_SYMBOL_ENTRIES "")
foreach(data_file ${GOLDEN_DATA_FILES})
    get_filename_component(symbol "${data_file}" NAME_WE)
    string(APPEND GOLDEN_SYMBOL_ENTRIES "    \"${symbol}\",\n")
endforeach()
list(LENGTH GOLDEN_CLAIM_FILES GOLDEN_CLAIM_COUNT)
list(LENGTH GOLDEN_DATA_FILES GOLDEN_SYMBOL_COUNT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/golden_claims.h"
"/* Generated by examples/c/CMakeLists.txt -- do not edit. */
#ifndef WICKRA_VERIFY_GOLDEN_CLAIMS_H
#define WICKRA_VERIFY_GOLDEN_CLAIMS_H
#include <stddef.h>
#define GOLDEN_DIR \"${GOLDEN_DIR}\"
static const char *GOLDEN_CLAIMS[] = {
${GOLDEN_CLAIM_ENTRIES}    NULL
};
static const size_t GOLDEN_CLAIM_COUNT = ${GOLDEN_CLAIM_COUNT};
static const char *GOLDEN_SYMBOLS[] = {
${GOLDEN_SYMBOL_ENTRIES}    NULL
};
static const size_t GOLDEN_SYMBOL_COUNT = ${GOLDEN_SYMBOL_COUNT};
#endif
")

# The test, as opposed to the examples above: `verify` and `verifypp` print a
# verdict and exit zero on a refutation. This one asserts golden parity and
# operating-mode equivalence over the whole corpus.
add_executable(golden_test golden_test.c)
target_include_directories(golden_test PRIVATE
    "${WICKRA_VERIFY_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(golden_test PRIVATE "${WICKRA_VERIFY_LIB}")

# On Windows the runtime DLL must sit next to the executable (there is no rpath);
# without it the loader pops a modal "DLL not found" dialog that hangs ctest. Copy
# it next to each example after build. (Unix resolves the .so/.dylib via the
# linker's embedded path, so this is Windows-only.)
if(WIN32)
    foreach(example verify verifypp golden_test)
        add_custom_command(TARGET ${example} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${WICKRA_VERIFY_LIB_DIR}/wickra_verify.dll"
                "$<TARGET_FILE_DIR:${example}>"
            VERBATIM)
    endforeach()
endif()

enable_testing()
add_test(NAME verify COMMAND verify)
add_test(NAME verifypp COMMAND verifypp)
add_test(NAME golden_test COMMAND golden_test)
# A missing runtime dependency must fail fast, never block the runner for hours.
set_tests_properties(verify verifypp golden_test PROPERTIES TIMEOUT 60)
